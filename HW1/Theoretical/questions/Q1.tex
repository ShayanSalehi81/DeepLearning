\subsection*{بخش اول}

\subsection*{آ}
برای
$f:\mathbb{R}^n\to\mathbb{R}^m$،
$g:\mathbb{R}^m\to\mathbb{R}^k$
و
$h(x)=g(f(x))$،
طبق قاعده زنجیره‌ای برای ژاکوبین داریم:
\begin{flalign*}
J_h(x)
&= J_g\!\big(f(x)\big)\; J_f(x) &&
\end{flalign*}
که در آن
$J_f(x)\in\mathbb{R}^{m\times n}$،
$J_g(f(x))\in\mathbb{R}^{k\times m}$
و بنابراین
$J_h(x)\in\mathbb{R}^{k\times n}$.

\subsection*{ب}
فرض کنید
$f(x)=\mathrm{ReLU}(Ax+b)$
و
$g(z)=Cz$.

\begin{enumerate}
\item داریم:
\begin{flalign*}
h(x)=g(f(x))=C\,\mathrm{ReLU}(Ax+b) &&
\end{flalign*}

\item اگر
$u=Ax+b\in\mathbb{R}^m$،
آنگاه
$J_g(z)=C$
و
$J_f(x)=D(u)\,A$
که در آن
$D(u)$ یک ماتریس قطری است. پس:
\begin{flalign*}
J_h(x)
&=J_g(f(x))\,J_f(x)=C\,D(Ax+b)\,A &&
\end{flalign*}

\item مشتق \lr{ReLU} را می‌توان به صورت ماتریس قطری زیر نوشت:
\begin{flalign*}
D(u) &= \mathrm{diag}\big(r_1,\dots,r_m\big), \qquad
r_i=\frac{\partial\,\mathrm{ReLU}(u_i)}{\partial u_i}
=
\begin{cases}
1 & u_i>0\\
0 & u_i<0
\end{cases} &&
\end{flalign*}
(در نقطه $u_i=0$ مشتق کلاسیک تعریف نشده و می‌توان یکی از زیرگرادیان‌ها در بازه $[0,1]$ را انتخاب کرد.)
\end{enumerate}

\subsection*{ج}
فرض کنید
$f(x)=\mathrm{Softmax}(Ax+b)$
و
$g(z)=Cz$.
بگذارید
$z=Ax+b$
و
$s=\mathrm{Softmax}(z)$.

\begin{enumerate}
\item تابع ترکیبی:
\begin{flalign*}
h(x)=g(f(x))=C\,\mathrm{Softmax}(Ax+b)=C\,s &&
\end{flalign*}

\item برای محاسبه ژاکوبین \lr{Softmax}:
\begin{flalign*}
s_i &= \frac{e^{z_i}}{\sum_{t=1}^m e^{z_t}} \\
\frac{\partial s_i}{\partial z_j}
&=
\frac{\partial}{\partial z_j}\left(\frac{e^{z_i}}{\sum_{t} e^{z_t}}\right)
=
\frac{\delta_{ij}e^{z_i}\left(\sum_t e^{z_t}\right)-e^{z_i}e^{z_j}}{\left(\sum_t e^{z_t}\right)^2}
=
\frac{e^{z_i}}{\sum_t e^{z_t}}
\left(\delta_{ij}-\frac{e^{z_j}}{\sum_t e^{z_t}}\right) \\
&= s_i(\delta_{ij}-s_j) &&
\end{flalign*}
پس ژاکوبین به صورت ماتریسی نیز برابر است با:
\begin{flalign*}
J_{\mathrm{Softmax}}(z)
&= \mathrm{diag}(s) - s\,s^\top \in \mathbb{R}^{m\times m} &&
\end{flalign*}

\item با قاعده زنجیره‌ای:
\begin{flalign*}
J_h(x)
&= J_g(f(x))\,J_f(x)
= C \; J_{\mathrm{Softmax}}(Ax+b)\; A \\
&= C\Big(\mathrm{diag}(s)-s s^\top\Big)A,\qquad s=\mathrm{Softmax}(Ax+b) &&
\end{flalign*}
که اندازه آن
$k\times n$
است.
\end{enumerate}

\subsection*{بخش دوم}

\subsection*{آ}
چون
$z=Wx+b$،
ژاکوبین $z$ نسبت به $x$ برابر است با:
\begin{flalign*}
\frac{\partial z}{\partial x} = W \in \mathbb{R}^{m\times n} &&
\end{flalign*}

\subsection*{ب}
داریم
$L=\frac{1}{2}\|z-y\|^2=\frac{1}{2}(z-y)^\top(z-y)$،
پس:
\begin{flalign*}
\nabla_z L &= z-y \\
\nabla_b L &= \nabla_z L \cdot \frac{\partial z}{\partial b}
= (z-y)\cdot I_m
= z-y \in \mathbb{R}^{m} &&
\end{flalign*}

\subsection*{ج}
\begin{flalign*}
\nabla_x L
&= \left(\frac{\partial z}{\partial x}\right)^\top \nabla_z L
= W^\top (z-y)
\in \mathbb{R}^{n} &&
\end{flalign*}

\subsection*{د}
برای مشتق نسبت به $W$:
\begin{flalign*}
\nabla_W L
&= (z-y)\,x^\top
\in \mathbb{R}^{m\times n} &&
\end{flalign*}
