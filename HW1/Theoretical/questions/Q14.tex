\section*{پرسش ۱. مرور ریاضیات برای یادگیری عمیق (۱۵ + ۵ نمره)}

\subsection*{بخش اول}

فرض کنید
\begin{equation*}
    f: \mathbb{R}^n \to \mathbb{R}^m
\end{equation*}
و
\begin{equation*}
    g: \mathbb{R}^m \to \mathbb{R}^k
\end{equation*}
دو تابع مشتق‌پذیر باشند و تابع ترکیبی زیر تعریف شود:
\begin{equation*}
    h(x) = g(f(x))
\end{equation*}

\begin{itemize}
    \item[آ)] ماتریس ژاکوبین تابع $h(x)$ را بر حسب ماتریس‌های ژاکوبین $J_f(x)$ و $J_g(x)$ بدست آورید.
    
    \item[ب)] فرض کنید
    \begin{equation*}
        f(x) = \text{ReLU}(Ax + b), \quad g(z) = Cz
    \end{equation*}
    که در آن
    \begin{equation*}
        A \in \mathbb{R}^{m \times n}, \quad b \in \mathbb{R}^m, \quad C \in \mathbb{R}^{k \times m}
    \end{equation*}
    و تابع ReLU به صورت زیر تعریف می‌شود:
    \begin{equation*}
        \text{ReLU}(u_i) = \max(0, u_i)
    \end{equation*}
    \begin{enumerate}
        \item ترکیبی $h(x) = g(f(x))$ را بنویسید.
        \item ماتریس ژاکوبین $J_h(x)$ را محاسبه کنید.
        \item مشتق تابع ReLU را با استفاده از یک ماتریس قطری بیان کنید.
    \end{enumerate}

    \item[ج)] فرض کنید تابع $f$ برابر با تابع Softmax باشد. برای یک بردار $z \in \mathbb{R}^m$ تابع Softmax به صورت زیر تعریف می‌شود:
    \begin{equation*}
        \text{Softmax}(z)_i = \frac{e^{z_i}}{\sum_{j=1}^m e^{z_j}}
    \end{equation*}
    اکنون فرض کنید
    \begin{equation*}
        f(x) = \text{Softmax}(Ax + b), \quad g(z) = Cz
    \end{equation*}
    \begin{enumerate}
        \item تابع ترکیبی $h(x) = g(f(x))$ را بنویسید.
        \item ژاکوبین تابع Softmax یعنی $J_{\text{Softmax}}(z)$ را محاسبه کنید و نشان دهید که مؤلفه‌های آن به صورت زیر هستند:
        \begin{equation*}
            \frac{\partial s_i}{\partial z_j} = s_i(\delta_{ij} - s_j)
        \end{equation*}
        که در آن $s = \text{Softmax}(z)$ و $\delta_{ij}$ دلتای کرونکر است.
        \item با استفاده از قاعده زنجیره‌ای، ماتریس ژاکوبین $J_h(x)$ را برای این حالت محاسبه کنید.
    \end{enumerate}
\end{itemize}

\subsection*{بخش دوم}

فرض کنید
\begin{equation*}
    x \in \mathbb{R}^n, \quad y \in \mathbb{R}^m, \quad W \in \mathbb{R}^{m \times n}, \quad b \in \mathbb{R}^m
\end{equation*}
و تابع زیر تعریف شده باشد:
\begin{align*}
    z &= Wx + b \\
    L(W, b, x) &= \frac{1}{2}\|z - y\|^2
\end{align*}

\begin{itemize}
    \item[آ)] ژاکوبین بردار $z$ نسبت به بردار $x$ را محاسبه کنید:
    \begin{equation*}
        \frac{\partial z}{\partial x}
    \end{equation*}
    و اندازه‌ی این ماتریس را مشخص کنید.
    \item[ب)] گرادیان تابع ضرر را نسبت به بردار بایاس $b$ محاسبه کنید:
    \begin{equation*}
        \nabla_b L
    \end{equation*}
    \item[ج)] گرادیان تابع ضرر را نسبت به ورودی $x$ محاسبه کنید:
    \begin{equation*}
        \nabla_x L
    \end{equation*}
    \item[د)] گرادیان تابع ضرر را نسبت به ماتریس وزن $W$ محاسبه کنید:
    \begin{equation*}
        \nabla_W L
    \end{equation*}
\end{itemize}

\section*{پرسش ۲. پس انتشار خطا (۱۵ نمره)}

فرض کنید یک شبکه‌ی عصبی دولایه به صورت زیر داریم:
\begin{align*}
    z_1 &= W_1 x^{(i)} + b_1 \\
    a_1 &= \tanh(z_1) \\
    z_2 &= W_2 a_1 + b_2 \\
    \hat{y}^{(i)} &= z_2 \\
    L^{(i)} &= \frac{1}{2} (\hat{y}^{(i)} - y^{(i)})^2 \\
    J &= \frac{1}{m} \sum_{i=1}^m L^{(i)}
\end{align*}

در این‌جا، $x^{(i)}$ نشان‌دهنده‌ی یک نمونه‌ی ورودی منفرد با ابعاد $D_x \times 1$ است. همچنین $y^{(i)}$ برچسب خروجی (هدف) متناظر با این نمونه بوده و یک اسکالر است. تعداد کل نمونه‌های داده برابر با $m$ است. فرض می‌کنیم لایه‌ی مخفی دارای $D_{a1}$ نود باشد؛ بنابراین $z_1$ دارای ابعاد $D_{a1} \times 1$ خواهد بود.

\textbf{نکته:} در بخش مربوط به پیاده‌سازی برداری (vectorized)، منظور این است که کل داده‌ها، \textbf{یعنی هر $m$ نمونه، به‌صورت همزمان} به شبکه داده می‌شوند.

\begin{enumerate}
    \item ابعاد ماتریس‌ها و بردارهای $W_1$، $b_1$، $W_2$ و $b_2$ را مشخص کنید. اگر شبکه را روی چندین نمونه به‌صورت برداری (vectorized) پیاده‌سازی کنیم، ابعاد وزن‌ها و بایاس‌ها چه تغییری می‌کنند؟ در حالت برداری، ابعاد $X$ و $Y$ چگونه خواهند بود؟
    \item $\frac{\partial J}{\partial \hat{y}^{(i)}}$ را محاسبه کنید و آن را با $\delta_1^{(i)}$ نشان دهید.
    \item $\frac{\partial \hat{y}^{(i)}}{\partial z_2}$ را محاسبه کنید و آن را با $\delta_2^{(i)}$ نشان دهید.
    \item $\frac{\partial z_2}{\partial a_1}$ را محاسبه کنید و آن را با $\delta_3^{(i)}$ نشان دهید.
    \item $\frac{\partial a_1}{\partial z_1}$ را محاسبه کنید و آن را با $\delta_4^{(i)}$ نشان دهید.
    \item $\frac{\partial z_1}{\partial W_1}$ را محاسبه کنید و آن را با $\delta_5^{(i)}$ نشان دهید.
    \item $\frac{\partial J}{\partial W_1}$ را محاسبه کنید. می‌توانید از نتایج قسمت‌های قبل استفاده کنید. \textbf{راهنما:} به ابعاد ماتریس‌ها دقت کنید.
\end{enumerate}

\section*{پرسش ۳. روش‌های بهینه‌سازی مرتبه اول (۲۵ نمره)}

در این مسئله، تکامل الگوریتم‌های بهینه‌سازی مرتبه اول را بررسی می‌کنیم. هدف، تحلیل رفتار این الگوریتم‌ها برای کمینه‌سازی تابع محدب زیر است:
\begin{equation*}
    f(x, y) = 50x^2 + y^2
\end{equation*}
که در آن $x, y \in \mathbb{R}$ است. این تابع دارای انحنای بسیار زیاد در راستای محور $x$ و انحنای کم در راستای محور $y$ است، بنابراین یک مثال مناسب برای بررسی چالش‌های بهینه‌سازی محسوب می‌شود.

\begin{itemize}
    \item \textbf{الگوریتم Gradient Descent}
    
    الگوریتم گرادیان کاهشی پارامترها را در جهت بیشترین کاهش تابع هدف به‌روزرسانی می‌کند:
    \begin{equation*}
        x_{k+1} = x_k - \eta \nabla f(x_k)
    \end{equation*}
    که در آن $\eta > 0$ نرخ یادگیری است.
    \begin{itemize}
        \item[(الف)] گرادیان تابع داده شده $\nabla f(x, y)$ را محاسبه کنید.
        \item[(ب)] چرا استفاده از یک نرخ یادگیری ثابت $\eta$ برای کمینه‌سازی این تابع، با توجه به تفاوت انحنای آن در راستای محورهای $x$ و $y$، می‌تواند مشکل‌ساز باشد؟
        \item[(ج)] با فرض $\eta = 0.1$ و نقطه اولیه $(x_0, y_0) = (1, 1)$، دو تکرار اول الگوریتم گرادیان کاهشی را محاسبه کنید. آیا این نتایج نشان‌دهنده مشکلات ذکرشده در بخش (ب) هستند؟
    \end{itemize}

    \item \textbf{الگوریتم Adaptive Learning Rates (AdaGrad)}
    
    الگوریتم AdaGrad نرخ یادگیری هر پارامتر را بر اساس تاریخچه گرادیان‌های آن تنظیم می‌کند:
    \begin{align*}
        \eta_i^{(k)} &= \frac{\eta}{\sqrt{\sum_{j=1}^k (g_i^{(j)})^2} + \epsilon} \\
        x_{i, k+1} &= x_{i, k} - \eta_i^{(k)} g_i^{(k)}
    \end{align*}
    که در آن $g_i^{(j)}$ مؤلفه $i$-ام گرادیان در گام $j$ است و $\epsilon > 0$ برای جلوگیری از تقسیم بر صفر است.
    \begin{itemize}
        \item[(الف)] چرا تنظیم نرخ یادگیری به‌صورت جداگانه توسط AdaGrad برای هر پارامتر در تابع $f(x, y)$ مفید است؟
        \item[(ب)] با $\eta = 0.1$، نرخ‌های یادگیری مؤثر $\eta_x^{(1)}$ و $\eta_y^{(1)}$ را بعد از اولین تکرار محاسبه کنید (با فرض $\epsilon = 10^{-8}$).
        \item[(ج)] نرخ یادگیری در طول زمان چه تغییری می‌کند و چه اثری دارد؟
    \end{itemize}

    \item \textbf{الگوریتم RMSProp}
    
    الگوریتم RMSProp نسخه بهبودیافته AdaGrad است که به جای جمع کل گرادیان‌ها، از میانگین متحرک نمایی آن‌ها استفاده می‌کند:
    \begin{align*}
        s_{k+1} &= \gamma s_k + (1 - \gamma) g_k^2 \\
        x_{k+1} &= x_k - \frac{\eta}{\sqrt{s_{k+1}} + \epsilon} g_k
    \end{align*}
    که در آن $s_k$ میانگین متحرک نمایی مربع گرادیان‌ها و $\gamma \in [0, 1)$ ضریب کاهش است.
    \begin{itemize}
        \item[(الف)] استفاده از میانگین متحرک نمایی مربع گرادیان‌ها چه تفاوتی در مقایسه با AdaGrad ایجاد می‌کند؟ یک مزیت واضح RMSProp نسبت به AdaGrad را بیان کنید.
        \item[(ب)] نقش پارامتر $\gamma$ در RMSProp را با نقش $\beta$ در Momentum مقایسه کنید. هرکدام چه چیزی را کنترل می‌کنند؟
    \end{itemize}

    \item \textbf{الگوریتم Adam}
    
    Adam ترکیبی از Momentum و AdaGrad است:
    \begin{align*}
        m_{k+1} &= \beta_1 m_k + (1 - \beta_1) g_k \\
        v_{k+1} &= \beta_2 v_k + (1 - \beta_2) g_k^2 \\
        \hat{m}_{k+1} &= \frac{m_{k+1}}{1 - \beta_1^k}, \quad \hat{v}_{k+1} = \frac{v_{k+1}}{1 - \beta_2^k} \\
        x_{k+1} &= x_k - \frac{\eta}{\sqrt{\hat{v}_{k+1}} + \epsilon} \hat{m}_{k+1}
    \end{align*}
    \begin{itemize}
        \item[(الف)] بر اساس روابط بالا، مشخص کنید که کدام یک از متغیرهای $m_k$ و $v_k$ مربوط به Momentum و کدام مربوط به AdaGrad می‌باشند.
        \item[(ب)] چرا در الگوریتم Adam نیاز به تصحیح بایاس برای $m_k$ و $v_k$ وجود دارد؟ اگر این تصحیح انجام نشود، در چند تکرار ابتدایی چه مشکلی رخ می‌دهد؟
        \item[(ج)] برای تابع $f(x, y) = 50x^2 + y^2$ نتیجه‌گیری کنید که چرا انتظار می‌رود Adam عملکرد بهتری نسبت به گرادیان کاهشی ساده داشته باشد؟
    \end{itemize}
\end{itemize}

\section*{پرسش ۴. بهینه‌سازی مرتبه دوم (۱۵+۵ نمره)}

در این پرسش، قصد داریم تابع ضرر $E$ (با فرض داشتن شرایط لازم) را کمینه سازیم. فرض کنید $w^*$ یک نقطهٔ بهینه است به‌طوری که داریم $\nabla E(w^*) = 0$ و ماتریس هسیان $\nabla^2 E(w^*)$ مثبت‌معین است. با فرض بودن در همسایگی به اندازهٔ کافی کوچک $w^*$، طبق روش نیوتون داریم:
\begin{equation*}
w^{(k+1)} = w^{(k)} + p^{(k)}
\end{equation*}
که در آن $p^{(k)} := - \nabla^2 E(w^{(k)})^{-1} \cdot \nabla E(w^{(k)})$ است.

\textbf{(الف)}
نشان دهید تساوی زیر برقرار است:
\begin{equation}
\|\nabla E(w^{(k+1)})\| = \|\nabla E(w^{(k+1)}) - \nabla E(w^{(k)}) - \nabla^2 E(w^{(k)})p^{(k)}\|.
\end{equation}

\textbf{(ب)}
طبق قضیهٔ تیلور، برای هر تابعی مانند $f : \mathbb{R}^d \to \mathbb{R}$ که دو مرتبه مشتق‌پذیر باشد و مشتق آن پیوسته باشد و به ازای هر $p \in \mathbb{R}^d$ و مقدار $t \in (0,1)$ داریم:
\begin{equation}
\nabla f(x+p) = \nabla f(x) + \int_0^1 \nabla^2 f(x+tp)p\, dt.
\end{equation}

با استفاده از روابط (۱) و (۲) و خواص عملگرهای نُرم و انتگرال، نامساوی زیر را اثبات کنید.
\begin{equation}
\|\nabla E(w^{(k+1)})\| \le \int_0^1 \|\nabla^2 E(w^{(k)}+tp^{(k)}) - \nabla^2 E(w^{(k)})\| \cdot \|p^{(k)}\|\, dt
\end{equation}

\textbf{(ج)}
فرض می‌کنیم $\nabla^2 E(\cdot)$ یک تابع $L$-لیپشیتز است؛ یعنی به ازای هر $u$ و $v$ داریم:
\begin{equation*}
\|\nabla^2 E(u) - \nabla^2 E(v)\| \le L \cdot \|u - v\|.
\end{equation*}

با استفاده از رابطهٔ (۳) و فرض فوق، نشان دهید:
\begin{equation*}
\|\nabla E(w^{(k+1)})\| \le \frac{1}{2} L \|\nabla^2 E(w^{(k)})^{-1}\|^2 \cdot \|\nabla E(w^{(k)})\|^2
\end{equation*}

\textbf{(د)}
با توجه به معکوس‌پذیری $\nabla^2 E(w^*)$، وجود دارد شعاع $r > 0$ طوری که داریم $\|\nabla^2 E(w^{(k)})^{-1}\| \le 2 \|\nabla^2 E(w^*)^{-1}\|$ به ازای تمام $w^{(k)}$ هایی که حداکثر فاصله $r$ دارند یعنی $\|w^{(k)} - w^*\| \le r$. با استفاده از این خاصیت و بخش‌های قبل، نتیجه گرفت:
\begin{equation}
\|\nabla E(w^{(k+1)})\| \le 2 L \|\nabla^2 E(w^*)^{-1}\|^2 \cdot \|\nabla E(w^{(k)})\|^2
\end{equation}

با توجه به آنکه رابطهٔ (۴) بیانگر تغییرات نُرم گرادیان در هر گام نسبت به گام قبلی است، چه تفسیری می‌توان در مورد نقطهٔ هم‌گرایی و نرخ هم‌گرایی برای نُرم گرادیان‌ها ارائه کرد؟

\section*{پرسش ۵. خواص پایه‌ای توابع لیپشیتز (۱۵ نمره)}

در این بخش، چند خاصیت بنیادی توابع لیپشیتز را که به عنوان ابزارهای ریاضی برای تحلیل شبکه‌های عصبی به کار می‌روند، اثبات و بررسی می‌کنیم. ابتدا به تعریف لیپشیتز می‌پردازیم.
اگر برای تابعی نظیر $f : \mathbb{R}^n \to \mathbb{R}$ داشته باشیم:
\begin{equation*}
\|f(x_1) - f(x_2)\| \le L\|x_1 - x_2\|, \quad \forall x_1, x_2 \in \mathbb{R}^n
\end{equation*}
می‌گوییم این تابع $L$-Lipschitz است. برای درک بهتر این مفهوم $x_1 = x_2 + \delta$ را در نظر داشته باشید. اکنون می‌توان ادعا کرد که برای هر $x_2$ و به ازای هر $\delta$ دلخواه داریم:
\begin{equation*}
\|f(x_2+\delta) - f(x_2)\| \le L\|\delta\|
\end{equation*}
یعنی اگر به ازای هر مقدار ورودی نظیر $x$، آن را به اندازه $\delta$ تغییر دهیم، میزان تغییرات در خروجی تابع کمتر از $L\|\delta\|$ خواهیم داشت و به بیانی یک کران بالا برای میزان تغییرات تعریف کردیم.
می‌دانیم که در هر لایه از یک شبکه عصبی داریم:
\begin{equation*}
x^{(\ell+1)} = h^{(\ell)}(W^{(\ell)}x^{(\ell)})
\end{equation*}
که در اینجا $h(\cdot)$ یک تابع فعال‌سازی دلخواه است. حال به پرسش‌های زیر پاسخ دهید.

\textbf{۱. جمع توابع لیپشیتز:} ثابت کنید اگر $f : \mathbb{R}^n \to \mathbb{R}^m$ یک تابع $L_f$-لیپشیتز و $g : \mathbb{R}^n \to \mathbb{R}^m$ یک تابع $L_g$-لیپشیتز باشد، آنگاه تابع مجموع $h(x) = f(x) + g(x)$ نیز لیپشیتز است. کران بالای ثابت لیپشیتز آن ($L_h$) چیست؟

\textbf{۲. ترکیب توابع لیپشیتز:}
\begin{itemize}
    \item[\textbf{(الف)}] \textbf{اثبات خاصیت ترکیب:} ثابت کنید اگر $g : \mathbb{R}^n \to \mathbb{R}^m$ یک تابع $L_g$-لیپشیتز و $f : \mathbb{R}^m \to \mathbb{R}^k$ یک تابع $L_f$-لیپشیتز باشد، آنگاه تابع مرکب $h(x) = f(g(x))$ نیز یک تابع لیپشیتز است. کران بالای ثابت لیپشیتز آن ($L_h$) چیست؟
    
    \item[\textbf{(ب)}] \textbf{کران ثابت لیپشیتز یک لایه:} یک لایه شبکه عصبی به صورت $f_k(x) = \sigma(W_k x)$ را در نظر بگیرید که در آن $\sigma$ یک تابع فعال‌سازی $L_\sigma$-لیپشیتز و $W_k$ ماتریس وزن است. با استفاده از خاصیت ترکیب توابع که در بند (الف) اثبات کردید، نشان دهید که ثابت لیپشیتز این لایه ($L_k$) توسط $L_k \le L_\sigma \cdot \|W_k\|_2$ کران‌دار می‌شود. (یادآوری: ثابت لیپشیتز تبدیل خطی $x \mapsto Wx$ برابر با نورم طیفی ماتریس، یعنی $\|W\|_2$ است).
\end{itemize}

\textbf{۳. ارتباط با مشتق:}
برای یک تابع مشتق‌پذیر $f : \mathbb{R} \to \mathbb{R}$، ثابت کنید که $f$ یک تابع $L$-لیپشیتز است اگر و تنها اگر برای تمام مقادیر $x$ داشته باشیم $|f'(x)| \le L$.

\section*{پرسش ۶. Regularization (۱۵ نمره)}

ماتریس طراحی $X$ (که سطر $i$-ام آن نقطهٔ نمونهٔ $x_i^T$ است) و یک بردار $n$-تایی از برچسب‌ها $y = [y_1, \dots, y_n]^T$ به شما داده شده است. برای سادگی، فرض کنید $X$ سفید (whitened) شده است، به طوری که $X^T X = nI$ باشد. جملهٔ بایاس یا بعد اضافی اضافه نکنید؛ برای ورودی صفر، خروجی همیشه صفر است. فرض کنید $x_{*i}$ ستون $i$-ام ماتریس $X$ را نشان می‌دهد.

\textbf{۱.} نشان دهید که تابع هزینه برای کمترین مربعات با تنظیم‌گر $\ell_1$ (Lasso)، یعنی:
\begin{equation*}
J_1(w) \triangleq \|Xw - y\|^2 + \lambda \|w\|_1 \quad \quad \text{(که در آن $\lambda > 0$)}
\end{equation*}
می‌تواند به صورت زیر بازنویسی شود:
\begin{equation*}
J_1(w) = \|y\|^2 + \sum_{i=1}^d f(x_{*i}, w_i)
\end{equation*}
که در آن $f(\cdot, \cdot)$ یک تابع مناسب است که آرگومان اول آن یک بردار و آرگومان دوم آن یک اسکالر است.

\textbf{۲.} با استفاده از پاسخ خود به بخش (۱)، شرایط لازم و کافی برای مولفهٔ $i$-ام بهینه‌ساز $w^*$ تابع $J_1(\cdot)$ را استخراج کنید به طوری که هر یک از سه خاصیت زیر را برآورده کند:
\begin{equation*}
w_i^* > 0, \quad w_i^* = 0, \quad w_i^* < 0
\end{equation*}

\textbf{۳.} برای بهینه‌ساز $w^\#$ تابع هزینهٔ کمترین مربعات با تنظیم‌گر $\ell_2$ (Ridge)، یعنی:
\begin{equation*}
J_2(w) \triangleq \|Xw - y\|^2 + \lambda \|w\|_2^2 \quad \quad \text{(که در آن $\lambda > 0$)}
\end{equation*}
یک شرط لازم و کافی برای $w_i^\# = 0$ استخراج کنید، که در آن $w_i^\#$ مولفهٔ $i$-ام بردار $w^\#$ است.

\textbf{۴.} یک بردار را تُنُک (Sparse) می‌نامیم اگر بیشتر مولفه‌های آن صفر باشند. بر اساس پاسخ‌های خود به بخش‌های (۲) و (۳)، کدام یک از بردارهای $w^*$ و $w^\#$ به احتمال زیاد تُنُک‌تر است؟ چرا؟
